  num int;
  i int:=0;
begin
 num:=&num;

  loop
    if num>0 then 
      dbms_output.put_line('posivtive');
      if i mod 2=1 then
        dbms_output.put_line('odd='||i);
      end if;
   else
      dbms_output.put_line('negitive');

  end loop;
end;
