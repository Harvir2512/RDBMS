DECLARE
    CURSOR C1 IS SELECT * FROM EMP;
    U_ENO INT ;
    U_ENAME VARCHAR(20);
    U_SAL FLOAT;
BEGIN
    OPEN C1;
    LOOP
         FETCH C1 INTO U_ENO,U_ENAME,U_SAL;
         EXIT WHEN C1%NOTFOUND;
         DBMS_OUTPUT.PUT_LINE('EMPLOYEE NUMBER='||U_ENO);
         DBMS_OUTPUT.PUT_LINE('EMPLOYEE NAME='||U_ENAME);
         DBMS_OUTPUT.PUT_LINE('EMPLOYEE SALARY='||U_SAL);
    END LOOP;
    CLOSE C1;
END;
